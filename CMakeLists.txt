cmake_minimum_required(VERSION 3.16)
project(console_sequence LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This project uses Windows console APIs; use a Windows C++ toolchain.")
endif()
add_executable(console-sequence Sequence.cpp)
target_compile_features(console-sequence PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(console-sequence PRIVATE /W4 /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(input_test tests/input_test.cpp)
  target_compile_features(input_test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(input_test PRIVATE /utf-8 /UNDEBUG)
  else()
    target_compile_options(input_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME input_test COMMAND input_test)
  add_executable(rules_test tests/rules_test.cpp)
  target_compile_features(rules_test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(rules_test PRIVATE /utf-8 /UNDEBUG)
  else()
    target_compile_options(rules_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME rules_test COMMAND rules_test)
endif()
